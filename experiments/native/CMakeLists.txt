cmake_minimum_required(VERSION 3.19)
project(blind_combat_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LIGHTSPEED "${CMAKE_CURRENT_SOURCE_DIR}/../../.tools/sts_lightspeed" CACHE PATH "Pinned upstream checkout")
file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS "${LIGHTSPEED}/src/*.cpp")
add_executable(blind-combat-lab blind_combat_lab.cpp ${ENGINE_SOURCES})
target_include_directories(blind-combat-lab PRIVATE "${LIGHTSPEED}/include")
target_include_directories(blind-combat-lab SYSTEM PRIVATE "${LIGHTSPEED}/json/include")
